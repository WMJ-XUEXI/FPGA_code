  wire [95:0] display_rgb;
  genvar pixel;
  generate for (pixel = 0; pixel < 4; pixel = pixel + 1) begin : display_pixels
      wire [7:0] r_in = awb_O_tdata[pixel*24+16 +: 8];
      wire [7:0] g_in = awb_O_tdata[pixel*24+8 +: 8];
      wire [7:0] b_in = awb_O_tdata[pixel*24 +: 8];
      wire [7:0] r = (r_in > DISPLAY_BLACK_TRIM) ? r_in - DISPLAY_BLACK_TRIM : 8'd0;
      wire [7:0] g = (g_in > DISPLAY_BLACK_TRIM) ? g_in - DISPLAY_BLACK_TRIM : 8'd0;
      wire [7:0] b = (b_in > DISPLAY_BLACK_TRIM) ? b_in - DISPLAY_BLACK_TRIM : 8'd0;
      wire [7:0] rg_peak = (r > g) ? r : g;
      wire [7:0] peak_value = (rg_peak > b) ? rg_peak : b;
      wire [15:0] gain = display_gain(peak_value);
      wire [23:0] r_scaled = r * gain;
      wire [23:0] g_scaled = g * gain;
      // Reduce residual yellow/green cast: green trim = 61/64 (-4.6875%).
      wire [23:0] g_balanced = g_scaled - (g_scaled >> 5) - (g_scaled >> 6);
      wire [23:0] b_scaled = b * gain;
      assign display_rgb[pixel*24+16 +: 8] = (|r_scaled[23:16]) ? 8'd255 : r_scaled[15:8];
      assign display_rgb[pixel*24+8 +: 8] = (|g_balanced[23:16]) ? 8'd255 : g_balanced[15:8];
      assign display_rgb[pixel*24 +: 8] = (|b_scaled[23:16]) ? 8'd255 : b_scaled[15:8];
  end endgenerate
